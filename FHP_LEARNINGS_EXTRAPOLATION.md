class DROStateSpace:
    """Full state-space model of a Dynamic Resonant Object"""

    def __init__(
        self,
        omega_0: float,      # Natural frequency (identity)
        zeta: float,         # Damping ratio (metabolism)
        A_0: float = 1.0,    # Initial amplitude
        phi_0: float = 0.0   # Initial phase
    ):

        self.omega_d = omega_0 * np.sqrt(1 - zeta**2)  # Damped frequency
        self.zeta = zeta
        self.gamma = zeta * omega_0  # Decay constant

        # State vector: [amplitude, phase, velocity]
        self.state = np.array([A_0, phi_0, 0.0])
        self.t = 0.0

        # Interaction history (for phase-lock calculation)
        self.interaction_times = []
        self.neighbors = []  # Other DROs in resonance range

    def evolve(self, dt: float, F_ext: float = 0.0) -> np.array:
        """Evolve state by dt seconds with external force F_ext"""
        A, phi, v = self.state

        # Amplitude decay
        A_new = A * np.exp(-self.gamma * dt)

        # Phase advance
        phi_new = (phi + self.omega_d * dt) % (2 * np.pi)

        # External forcing (impulse response)
        if F_ext > 0:
            # Impulse adds to amplitude, efficiency depends on phase
            phase_efficiency = 0.5 * (1 + np.cos(phi))  # Max at phi=0

        A_new += resonance_bonus * dt

        A_new = np.clip(A_new, 0.0, 2.0)  # Max 2x baseline

        self.state = np.array([A_new, phi_new, (A_new - A) / dt])
        self.t += dt

pub struct DRO {
    pub omega_0: f64,           // Natural frequency (immutable)
    pub zeta: f64,              // Damping ratio
    pub last_amplitude: f64,    // Amplitude at last interaction
    pub last_timestamp: i64,    // Unix timestamp of last update
    pub last_phase: f64,        // Phase at last interaction

    pub metadata_uri: String,   // Points to shader parameters

impl DRO {
    pub fn current_amplitude(&self, now: i64) -> f64 {
        let dt = (now - self.last_timestamp) as f64;
        let gamma = self.zeta * self.omega_0;
        self.last_amplitude * (-gamma * dt).exp()
    }

    pub fn current_phase(&self, now: i64) -> f64 {
        let dt = (now - self.last_timestamp) as f64;
        let omega_d = self.omega_0 * (1.0 - self.zeta.powi(2)).sqrt();
        (self.last_phase + omega_d * dt) % (2.0 * std::f64::consts::PI)
    }

    pub fn pump(&mut self, now: i64, force: f64) -> Result<()> {
        // Calculate current state (lazy)
        let current_amp = self.current_amplitude(now);
        let current_phase = self.current_phase(now);

        // Phase-dependent efficiency

        // Update state
        self.last_amplitude = (current_amp + amp_boost).min(2.0);
        self.last_phase = current_phase;
        self.last_timestamp = now;

        Ok(())
    }

    pub fn is_decoherent(&self, now: i64) -> bool {
        self.current_amplitude(now) < 0.001
    }
